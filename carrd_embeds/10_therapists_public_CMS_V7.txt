<script>

(function(){

var status =
document.getElementById(
'sepsyg-map-direct-status'
);

var root =
document.getElementById(
'sepsyg-map-direct-root'
);

/* ========================================
ΒΡΙΣΚΕΙ ΤΟ ΜΕΓΑΛΟ SVG ΤΟΥ ΧΑΡΤΗ
======================================== */

function findMainMapSvg(){

if(!root) return null;

var svgs =
Array.from(
root.querySelectorAll('svg')
);

var biggestSvg = null;

var biggestArea = 0;

svgs.forEach(function(svg){

var rect =
svg.getBoundingClientRect();

var area =
rect.width *
rect.height;

/*
Αγνοούμε μικρά SVG icons
και κρατάμε μόνο μεγάλο SVG.
*/

if(
rect.width > 280 &&
rect.height > 180 &&
area > biggestArea
){

biggestArea = area;

biggestSvg = svg;

}

});

return biggestSvg;

}

/* ========================================
ΒΡΙΣΚΕΙ ΤΟ ΚΕΙΜΕΝΟ
"Επίλεξε μια πινέζα..."
======================================== */

function findMapTextElement(){

if(!root) return null;

var all =
Array.from(
root.querySelectorAll('*')
);

for(
var i = 0;
i < all.length;
i++
){

var el = all[i];

var text =
(el.textContent || '')
.trim();

/*
Θέλουμε μικρό στοιχείο που
περιέχει το συγκεκριμένο κείμενο,
όχι ολόκληρο το application.
*/

if(
text.indexOf(
'Επίλεξε μια πινέζα'
) !== -1 ||

text.indexOf(
'Επίλεξε μία πινέζα'
) !== -1
){

if(
el.children.length <= 3
){

return el;

}

}

}

return null;

}

/* ========================================
LOWEST COMMON ANCESTOR
======================================== */

function commonAncestor(
elementA,
elementB
){

if(
!elementA ||
!elementB
){

return null;

}

var ancestors =
new Set();

var current =
elementA;

while(current){

ancestors.add(current);

current =
current.parentElement;

}

current =
elementB;

while(current){

if(
ancestors.has(current)
){

return current;

}

current =
current.parentElement;

}

return null;

}

/* ========================================
ΒΡΙΣΚΕΙ ΜΟΝΟ ΤΟ ΕΠΑΝΩ PANEL

Χάρτης + δεξιά καρτέλα
======================================== */

function findMapPanel(){

var svg =
findMainMapSvg();

var text =
findMapTextElement();

if(
!svg ||
!text
){

return null;

}

var panel =
commonAncestor(
svg,
text
);

if(
!panel ||
panel === root
){

return null;

}

/*
Ανεβαίνουμε λίγο προς τα πάνω,
ώστε να πιάσουμε ολόκληρο
το πλαίσιο Χάρτης + Καρτέλα.

Σταματάμε μόλις ο parent
αρχίσει να περιέχει το section
"Όλοι οι Θεραπευτές".
*/

while(
panel.parentElement &&
panel.parentElement !== root
){

var parent =
panel.parentElement;

var parentText =
parent.textContent || '';

if(

parentText.indexOf(
'Όλοι οι Θεραπευτές'
) !== -1

){

break;

}

panel = parent;

}

return panel;

}

/* ========================================
ΜΙΚΡΑΙΝΕΙ ΑΝΑΛΟΓΙΚΑ ΤΟ PANEL
======================================== */

function compactMapPanel(){

var panel =
findMapPanel();

if(!panel){

return;

}

/*
Αφαιρούμε πιθανή παλιά κλάση
από άλλο element.
*/

var oldPanels =
root.querySelectorAll(
'.sepsyg-map-panel-compact'
);

oldPanels.forEach(
function(oldPanel){

if(
oldPanel !== panel
){

oldPanel.classList.remove(
'sepsyg-map-panel-compact'
);

oldPanel.style.removeProperty(
'--sepsyg-map-scale'
);

oldPanel.style.removeProperty(
'margin-bottom'
);

}

});

panel.classList.add(
'sepsyg-map-panel-compact'
);

/*
Desktop: 94%
Mobile: 98%

Άρα το desktop panel γίνεται
ελαφρώς μικρότερο,
χωρίς crop.
*/

var scale =
window.innerWidth <= 720
? .98
: .94;

panel.style.setProperty(
'--sepsyg-map-scale',
scale
);

/*
Το transform μικραίνει οπτικά
το panel αλλά το browser εξακολουθεί
να κρατά το αρχικό ύψος.

Με αυτό αφαιρούμε μόνο τον
περιττό κενό χώρο από κάτω.

ΔΕΝ κόβουμε περιεχόμενο.
*/

requestAnimationFrame(
function(){

var originalHeight =
panel.offsetHeight;

var difference =
originalHeight *
(1 - scale);

panel.style.marginBottom =
(-Math.round(difference))
+ 'px';

}
);

}

/* ========================================
OBSERVER

Το embed.js δημιουργεί το app
μετά τη φόρτωση.
======================================== */

if(root){

var observer =
new MutationObserver(
function(){

compactMapPanel();

}
);

observer.observe(

root,

{

childList:true,
subtree:true

}

);

}

/* ========================================
RESIZE
======================================== */

window.addEventListener(

'resize',

function(){

compactMapPanel();

}

);

/* ========================================
EMBED LOADED
======================================== */

window.sepsygMapEmbedLoaded =
function(){

if(status){

status.style.display =
'none';

}

/*
Το τρέχουμε μερικές φορές
γιατί η εφαρμογή μπορεί να
ολοκληρώνει το render ασύγχρονα.
*/

setTimeout(
compactMapPanel,
100
);

setTimeout(
compactMapPanel,
400
);

setTimeout(
compactMapPanel,
900
);

setTimeout(
compactMapPanel,
1600
);

};

/* ========================================
EMBED FAILED
======================================== */

window.sepsygMapEmbedFailed =
function(){

if(!status) return;

status.classList.add(
'is-error'
);

status.textContent =

'Ο χάρτης δεν φορτώθηκε από το Vercel. Άνοιξε το /embed.js για έλεγχο.';

};

})();

</script>
