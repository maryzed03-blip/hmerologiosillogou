<script>
(function () {
const root = document.getElementById('sepsyg-menu-v3');
if (!root || root.dataset.accordionReady === '1') return;
root.dataset.accordionReady = '1';

const desktopDrops = Array.from(root.querySelectorAll('.sg-desktop-nav .sg-drop'));
const mobileSubs = Array.from(root.querySelectorAll('.sg-mobile-panel .sg-mobile-sub'));
const mobileMenu = root.querySelector('.sg-mobile-menu');

// Desktop: μόνο ένα dropdown ανοιχτό κάθε φορά.
desktopDrops.forEach(function (drop) {
drop.addEventListener('toggle', function () {
if (!drop.open) return;
desktopDrops.forEach(function (other) {
if (other !== drop) other.removeAttribute('open');
});
});
});

// Mobile: μόνο μία υποενότητα ανοιχτή κάθε φορά.
mobileSubs.forEach(function (sub) {
sub.addEventListener('toggle', function () {
if (!sub.open) return;
mobileSubs.forEach(function (other) {
if (other !== sub) other.removeAttribute('open');
});
});
});

// Αν πατηθεί κανονικό link, κλείνουν όσα dropdowns ήταν ανοιχτά.
root.querySelectorAll('.sg-desktop-nav a').forEach(function (link) {
link.addEventListener('click', function () {
desktopDrops.forEach(function (drop) {
drop.removeAttribute('open');
});
});
});

// Στο κινητό, μετά την επιλογή link κλείνει και όλο το hamburger menu.
root.querySelectorAll('.sg-mobile-panel a').forEach(function (link) {
link.addEventListener('click', function () {
mobileSubs.forEach(function (sub) {
sub.removeAttribute('open');
});
if (mobileMenu) mobileMenu.removeAttribute('open');
});
});

// Κλικ έξω από το menu = κλείσιμο όλων.
document.addEventListener('click', function (event) {
if (root.contains(event.target)) return;

desktopDrops.forEach(function (drop) {
drop.removeAttribute('open');
});

mobileSubs.forEach(function (sub) {
sub.removeAttribute('open');
});

if (mobileMenu) mobileMenu.removeAttribute('open');
});
})();
</script>
